### 2.2 核心表结构设计

#### 2.2.1 用户体系

```sql
-- 机构表
CREATE TABLE institutions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    status VARCHAR(20) DEFAULT 'pending',  -- pending, active, suspended
    principal_id UUID REFERENCES users(id),
    logo_url VARCHAR(500),
    contact_phone VARCHAR(20),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- 用户表
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    openid VARCHAR(128),  -- 微信openid
    unionid VARCHAR(128), -- 微信unionid
    phone VARCHAR(20) ENCRYPTED,
    password_hash VARCHAR(255),
    nickname VARCHAR(100),
    avatar_url VARCHAR(500),
    role VARCHAR(20) NOT NULL,  -- student, parent, teacher, principal, admin
    institution_id UUID REFERENCES institutions(id),
    age_range VARCHAR(20),  -- 用于学生
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- 家长-学生绑定表
CREATE TABLE parent_student (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    parent_id UUID REFERENCES users(id) ON DELETE CASCADE,
    student_id UUID REFERENCES users(id) ON DELETE CASCADE,
    relation VARCHAR(20),  -- father, mother, guardian
    created_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(parent_id, student_id)
);

-- 班级表
CREATE TABLE classes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    teacher_id UUID REFERENCES users(id),
    student_count INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT NOW()
);

-- 班级-学生关联表
CREATE TABLE class_students (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    student_id UUID REFERENCES users(id) ON DELETE CASCADE,
    joined_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(class_id, student_id)
);
```

## 四、核心组件设计

### 4.1 认证服务 (Auth Service)

```python
# services/auth.py

class AuthService:
    """认证服务"""

    def __init__(self, db: Session, redis: Redis):
        self.db = db
        self.redis = redis

    async def wechat_login(self, code: str, user_info: dict) -> TokenResponse:
        """微信一键登录"""
        # 1. 调用微信API获取openid
        openid = await self._get_wechat_openid(code)

        # 2. 查询或创建用户
        user = await self._get_or_create_user(openid, user_info)

        # 3. 生成JWT Token
        access_token = self._create_access_token(user)
        refresh_token = self._create_refresh_token(user)

        # 4. 存储Token到Redis（用于注销）
        await self._store_tokens(user.id, access_token, refresh_token)

        return TokenResponse(
            access_token=access_token,
            refresh_token=refresh_token,
            user=user
        )

    async def password_login(self, phone: str, password: str) -> TokenResponse:
        """账号密码登录（老师/校长/管理员）"""
        user = await self._verify_password(phone, password)
        if not user:
            raise AuthenticationError("手机号或密码错误")

        access_token = self._create_access_token(user)
        refresh_token = self._create_refresh_token(user)

        return TokenResponse(
            access_token=access_token,
            refresh_token=refresh_token,
            user=user
        )
```

### 4.2 语音评测服务 (Speech Service)

```python
# services/speech.py

class SpeechService:
    """语音评测服务（科大讯飞）"""

    def __init__(self, redis: Redis):
        self.redis = redis
        self.app_id = settings.XFY_APP_ID
        self.api_key = settings.XFY_API_KEY

    async def evaluate_pronunciation(
        self,
        audio_url: str,
        reference_text: str,
        student_id: str
    ) -> EvaluationResult:
        """跟读评分"""

        # 1. 从COS下载音频
        audio_data = await self._download_audio(audio_url)

        # 2. 调用讯飞语音评测API
        result = await self._xfy_evaluate(
            audio_data=audio_data,
            text=reference_text,
            scene="read_sentence"
        )

        # 3. 解析结果
        scores = self._parse_scores(result)

        # 4. 异步存储到数据库
        await self._save_record(student_id, audio_url, scores)

        return EvaluationResult(
            total=scores["total"],
            pronunciation=scores["pronunciation"],
            fluency=scores["fluency"],
            intonation=scores["intonation"],
            completeness=scores["completeness"]
        )

    async def submit_async_task(self, audio_url: str, text: str, task_id: str):
        """异步提交评测任务到Redis队列"""
        task_data = {
            "task_id": task_id,
            "audio_url": audio_url,
            "text": text,
            "created_at": datetime.now().isoformat()
        }
        await self.redis.lpush("speech_eval_queue", json.dumps(task_data))
```

### 4.3 AI出题服务 (AI Quiz Service)

```python
# services/ai_quiz.py

class AIQuizService:
    """AI智能出题服务"""

    def __init__(self, db: Session, redis: Redis, openai_client):
        self.db = db
        self.redis = redis
        self.openai = openai_client

    async def generate_quiz(self, book_id: str, student_id: str) -> QuizResult:
        """生成AI测验题"""

        # 1. 获取绘本内容
        book = await self._get_book(book_id)
        pages = await self._get_book_pages(book_id)

        # 2. 检查缓存（避免重复出题）
        cache_key = f"quiz:{book_id}:{student_id}"
        cached = await self.redis.get(cache_key)
        if cached:
            return json.loads(cached)

        # 3. 调用AI生成题目
        prompt = self._build_quiz_prompt(book, pages)
        response = await self.openai.chat.completions.create(
            model="gpt-4",
            messages=[
                {"role": "system", "content": "你是一位资深英语教育专家..."},
                {"role": "user", "content": prompt}
            ]
        )

        # 4. 解析AI响应
        questions = self._parse_questions(response.choices[0].message.content)

        # 5. 缓存题目
        result = QuizResult(quiz_id=uuid4(), questions=questions)
        await self.redis.setex(cache_key, 86400, result.json())  # 24小时缓存

        return result

    def _build_quiz_prompt(self, book, pages) -> str:
        """构建Prompt"""
        return f"""
请根据以下绘本内容生成3-5道测验题：

绘本标题：{book.title}
绘本级别：{book.level}
适读年龄：{book.age_range}

绘本内容：
{self._format_pages(pages)}

要求：
1. 生成选择题（4选1）和填词题
2. 难度与绘本级别匹配
3. 考察学生对绘本内容的理解
4. 返回JSON格式
"""
```

# 消费者示例
async def speech_eval_worker():
    """语音评测工作者"""
    while True:
        task = await redis.brpop(QUEUES["speech_eval"])
        task_data = json.loads(task)

        try:
            result = await speech_service.evaluate_pronunciation(
                audio_url=task_data["audio_url"],
                reference_text=task_data["reference_text"],
                student_id=task_data["student_id"]
            )

            # 存储结果
            await redis.setex(
                f"task:result:{task_data['task_id']}",
                3600,  # 1小时过期
                result.json()
            )

        except Exception as e:
            logger.error(f"评测任务失败: {e}")
